cmake_minimum_required(VERSION 3.18)
project(minifftdx VERSION 0.1.0 LANGUAGES CXX)

option(MINIFFTDX_BUILD_TESTS "Build the test suite" ON)
option(MINIFFTDX_BUILD_BENCH "Build the benchmarks" ON)
option(MINIFFTDX_BUILD_EXAMPLES "Build the examples" ON)
option(MINIFFTDX_FETCH_DEPS "Download GoogleTest / Google Benchmark if not installed" ON)
set(MINIFFTDX_ENABLE_CUDA "AUTO" CACHE STRING "CUDA backend: AUTO (only if nvcc is found), ON, OFF")
set_property(CACHE MINIFFTDX_ENABLE_CUDA PROPERTY STRINGS AUTO ON OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---- Optional CUDA ----------------------------------------------------------
set(MINIFFTDX_HAVE_CUDA OFF)
if(NOT MINIFFTDX_ENABLE_CUDA STREQUAL "OFF")
  include(CheckLanguage)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.24)
      set(CMAKE_CUDA_ARCHITECTURES native)
    else()
      set(CMAKE_CUDA_ARCHITECTURES 75)
    endif()
  endif()
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(MINIFFTDX_HAVE_CUDA ON)
  elseif(MINIFFTDX_ENABLE_CUDA STREQUAL "ON")
    message(FATAL_ERROR "MINIFFTDX_ENABLE_CUDA=ON but nvcc was not found")
  endif()
endif()
message(STATUS "MiniFFTDx CUDA backend: ${MINIFFTDX_HAVE_CUDA}")

# ---- The library (header-only) ---------------------------------------------
find_package(OpenMP COMPONENTS CXX)
add_library(minifftdx INTERFACE)
add_library(minifftdx::minifftdx ALIAS minifftdx)
target_include_directories(minifftdx INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(minifftdx INTERFACE cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(minifftdx INTERFACE OpenMP::OpenMP_CXX)
  message(STATUS "MiniFFTDx OpenMP: enabled")
else()
  message(STATUS "MiniFFTDx OpenMP: not found, CPU backend runs serially")
endif()
install(DIRECTORY include/ DESTINATION include)

if(MSVC)
  set(MFX_WARN /W4)
else()
  set(MFX_WARN -Wall -Wextra -Wpedantic)
endif()

# ---- Third-party ------------------------------------------------------------
include(FetchContent)
if(MINIFFTDX_BUILD_TESTS)
  find_package(GTest QUIET)
  if(NOT GTest_FOUND AND MINIFFTDX_FETCH_DEPS)
    FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
    set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
  endif()
endif()
if(MINIFFTDX_BUILD_BENCH)
  find_package(benchmark QUIET)
  if(NOT benchmark_FOUND AND MINIFFTDX_FETCH_DEPS)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_GTEST_TESTS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(benchmark URL https://github.com/google/benchmark/archive/refs/tags/v1.8.3.zip)
    FetchContent_MakeAvailable(benchmark)
  endif()
endif()

# ---- Examples ---------------------------------------------------------------
if(MINIFFTDX_BUILD_EXAMPLES)
  add_executable(fft_convolution examples/fft_convolution.cpp)
  target_link_libraries(fft_convolution PRIVATE minifftdx)
  target_compile_options(fft_convolution PRIVATE ${MFX_WARN})
endif()

# ---- Benchmarks -------------------------------------------------------------
if(MINIFFTDX_BUILD_BENCH AND (benchmark_FOUND OR TARGET benchmark::benchmark))
  add_executable(bench_fusion_cpu bench/bench_fusion.cpp)
  target_link_libraries(bench_fusion_cpu PRIVATE minifftdx benchmark::benchmark)
  if(MINIFFTDX_HAVE_CUDA)
    add_executable(bench_fusion_cuda bench/bench_fusion.cu)
    target_link_libraries(bench_fusion_cuda PRIVATE minifftdx benchmark::benchmark)
  endif()
endif()

# ---- Tests ------------------------------------------------------------------
if(MINIFFTDX_BUILD_TESTS AND (GTest_FOUND OR TARGET GTest::gtest))
  enable_testing()
  include(GoogleTest)

  add_library(mfx_test_main STATIC tests/test_main.cpp)
  target_link_libraries(mfx_test_main PUBLIC minifftdx GTest::gtest)
  target_include_directories(mfx_test_main PUBLIC tests)

  foreach(t test_descriptor test_accuracy test_cpu_backend test_fusion)
    add_executable(${t} tests/${t}.cpp)
    target_link_libraries(${t} PRIVATE mfx_test_main)
    target_compile_options(${t} PRIVATE ${MFX_WARN})
    gtest_discover_tests(${t} DISCOVERY_TIMEOUT 60)
  endforeach()

  # NumPy reference comparison (needs Python + numpy; skipped otherwise).
  find_package(Python3 COMPONENTS Interpreter QUIET)
  set(MFX_HAVE_NUMPY OFF)
  if(Python3_Interpreter_FOUND)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy" RESULT_VARIABLE _np_rc OUTPUT_QUIET ERROR_QUIET)
    if(_np_rc EQUAL 0)
      set(MFX_HAVE_NUMPY ON)
    endif()
  endif()
  add_executable(test_numpy_reference tests/test_numpy_reference.cpp)
  target_link_libraries(test_numpy_reference PRIVATE mfx_test_main)
  target_compile_options(test_numpy_reference PRIVATE ${MFX_WARN})
  if(MFX_HAVE_NUMPY)
    set(MFX_REFDIR ${CMAKE_BINARY_DIR}/numpy_ref)
    add_test(NAME numpy_generate_reference
             COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/gen_reference.py ${MFX_REFDIR})
    set_tests_properties(numpy_generate_reference PROPERTIES FIXTURES_SETUP numpy_ref)
    add_test(NAME numpy_reference COMMAND test_numpy_reference)
    set_tests_properties(numpy_reference PROPERTIES FIXTURES_REQUIRED numpy_ref
                         ENVIRONMENT "MINIFFTDX_REFDIR=${MFX_REFDIR}")
  else()
    message(STATUS "numpy not found: NumPy reference test disabled (pip install -r requirements.txt)")
  endif()

  # static_assert validation: each snippet in tests/compile_fail must FAIL to compile.
  file(GLOB MFX_BAD_CASES ${CMAKE_SOURCE_DIR}/tests/compile_fail/*.cpp)
  foreach(src ${MFX_BAD_CASES})
    get_filename_component(name ${src} NAME_WE)
    add_library(compile_fail_${name} OBJECT EXCLUDE_FROM_ALL ${src})
    target_link_libraries(compile_fail_${name} PRIVATE minifftdx)
    add_test(NAME compile_fail_${name}
             COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target compile_fail_${name} --config $<CONFIG>)
    set_tests_properties(compile_fail_${name} PROPERTIES WILL_FAIL TRUE)
  endforeach()

  if(MINIFFTDX_HAVE_CUDA)
    add_executable(test_cuda tests/test_cuda.cu)
    target_link_libraries(test_cuda PRIVATE mfx_test_main)
    gtest_discover_tests(test_cuda DISCOVERY_TIMEOUT 60)
  endif()
endif()
